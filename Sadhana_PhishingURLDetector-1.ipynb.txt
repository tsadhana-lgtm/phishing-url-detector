Legitimate URLs use well-known domains and normal paths; phishing URLs use common tricks (IP addresses, `@` symbols, long hyphenated lookalike domains, suspicious keywords, odd TLDs, no HTTPS). To use real data, replace this cell with `pd.read_csv('your_dataset.csv')` having columns `url` and `label` (0 = legitimate, 1 = phishing)."

    "legit_domains = [\"google.com\",\"wikipedia.org\",\"github.com\",\"microsoft.com\",\"amazon.in\",\"flipkart.com\",\n",
    "                 \"irctc.co.in\",\"nptel.ac.in\",\"ibm.com\",\"python.org\",\"stackoverflow.com\",\"linkedin.com\",\n",
    "                 \"youtube.com\",\"gov.in\",\"bbc.com\",\"apple.com\",\"swayam.gov.in\",\"coursera.org\"]\n",
    "legit_paths = [\"\", \"/\", \"/about\", \"/products/laptops\", \"/docs/getting-started\", \"/courses/python\",\n",
    "               \"/search?q=cyber+security\", \"/news/technology\", \"/login\", \"/help/contact\", \"/wiki/Computer_security\"]\n",
    "brands = [\"paypal\",\"sbi\",\"hdfcbank\",\"amazon\",\"netflix\",\"microsoft\",\"apple\",\"icici\",\"paytm\",\"google\"]\n",
    "words  = [\"login\",\"verify\",\"secure\",\"update\",\"account\",\"confirm\",\"wallet\",\"signin\",\"banking\",\"support\"]\n",
    "bad_tlds = [\"xyz\",\"top\",\"tk\",\"ml\",\"click\",\"info\",\"ru\",\"cf\",\"gq\"]\n",
    "\n",
    "def make_legit():\n",
    "    return random.choice([\"https://\",\"https://www.\",\"http://\"]) + random.choice(legit_domains) + random.choice(legit_paths)\n",
    "\n",
    "def make_phish():\n",
    "    kind = random.choice([\"ip\",\"at\",\"lookalike\",\"subdomain\",\"long\"])\n",
    "    b, w = random.choice(brands), random.choice(words)\n",
    "    tld = random.choice(bad_tlds)\n",
    "    if kind == \"ip\":\n",
    "        host = \".\".join(str(random.randint(1,255)) for _ in range(4))\n",
    "        return f\"http://{host}/{w}/{b}/index.php?id={random.randint(1000,99999)}\"\n",
    "    if kind == \"at\":\n",
    "        return f\"http://{b}.com@{random.randint(10,99)}.{random.randint(1,255)}.{random.randint(1,255)}.{random.randint(1,255)}/{w}\"\n",
    "    if kind == \"lookalike\":\n",
    "        return f\"http://{b}-{w}-{random.randint(10,999)}.{tld}/{w}.html\"\n",
    "    if kind == \"subdomain\":\n",
    "        return f\"http://{b}.{w}.secure-{random.randint(1,99)}.{tld}/account/{w}?redirect=true\"\n",
    "    return f\"http://{b}{w}-{random.choice(words)}-{random.choice(words)}.{tld}/{random.choice(words)}/{random.choice(words)}.php?user={random.randint(1,9999)}&token={random.randint(10**6,10**9)}\"\n",
    "\n",
    "rows = [(make_legit(),0) for _ in range(600)] + [(make_phish(),1) for _ in range(600)]\n",
    "df = pd.DataFrame(rows, columns=[\"url\",\"label\"]).sample(frac=1, random_state=42).reset_index(drop=True)\n",
    "print(df.shape); df.head(8)"

    "## 2. Feature extraction\n",
    "Each URL is converted into numeric features that security analysts commonly use as phishing indicators."
   ]
  },
  {
   "cell_type": "code",
   "metadata": {},

    "SUSPICIOUS = [\"login\",\"verify\",\"secure\",\"update\",\"account\",\"confirm\",\"wallet\",\"signin\",\"banking\",\"support\"]\n",
    "\n",
    "def extract_features(url):\n",
    "    p = urlparse(url)\n",
    "    host = p.netloc.split(\"@\")[-1]\n",
    "    return {\n",
    "        \"url_length\": len(url),\n",
    "        \"host_length\": len(host),\n",
    "        \"num_dots\": url.count(\".\"),\n",
    "        \"num_hyphens\": host.count(\"-\"),\n",
    "        \"num_digits\": sum(c.isdigit() for c in url),\n",
    "        \"num_special\": len(re.findall(r\"[?&=%@_]\", url)),\n",
    "        \"has_at_symbol\": int(\"@\" in url),\n",
    "        \"has_ip_host\": int(bool(re.fullmatch(r\"\\d{1,3}(\\.\\d{1,3}){3}\", host))),\n",
    "        \"uses_https\": int(p.scheme == \"https\"),\n",
    "        \"num_subdomains\": max(host.count(\".\") - 1, 0),\n",
    "        \"suspicious_words\": sum(w in url.lower() for w in SUSPICIOUS),\n",
    "        \"bad_tld\": int(host.split(\".\")[-1] in {\"xyz\",\"top\",\"tk\",\"ml\",\"click\",\"info\",\"ru\",\"cf\",\"gq\"}),\n",
    "        \"has_query\": int(bool(p.query)),\n",
    "        \"path_length\": len(p.path),\n",
    "    }\n",
    "\n",
    "X = pd.DataFrame([extract_features(u) for u in df[\"url\"]])\n",
    "y = df[\"label\"]\n",
    "X.head()"
   ]
